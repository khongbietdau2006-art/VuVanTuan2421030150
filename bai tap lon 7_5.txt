def la_so_nguyen_to(n):
    if n < 2: return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True
n = int(input("Nhập số lượng phần tử n (0 < n < 100): "))
a = []
for i in range(n):
    a.append(int(input(f"Nhập x{i+1}: ")))

tong_snt = sum(x for x in a if la_so_nguyen_to(x))

print(f"Tổng các số nguyên tố là: {tong_snt}")

# Kiểm tra điều kiện: là số lẻ và > 50
if tong_snt > 50 and tong_snt % 2 != 0:
    print("Tổng này là số lẻ và lớn hơn 50.")
else:
    print("Tổng này không thỏa mãn đồng thời là số lẻ và lớn hơn 50.")
